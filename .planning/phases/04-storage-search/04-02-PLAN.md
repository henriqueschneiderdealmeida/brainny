# Plan 04-02: GET /search semantic search endpoint

## Objective

Create `GET /search?q=<query>` endpoint:
- Auth: `Authorization: Bearer <SEARCH_TOKEN>` validated via `crypto.timingSafeEqual`
- Embed query text via `fastify.openai` (text-embedding-3-small)
- Query DB: cosine similarity on `messages.embedding` via pgvector, return top-20
- Response: `{ results: Array<{id, chatId, senderName, timestamp, text, score}> }`
- Wire into `src/index.ts`

## Context

- **Auth pattern:** Reuse `makeWebhookAuthHandler` approach (timingSafeEqual with byteLength guard). For search, the token comes from `Authorization: Bearer <token>` header (not a custom header).
- **Embedding call:** Use `fastify.openai.embeddings.create({ model: 'text-embedding-3-small', input: q })` — same client as enrich.ts
- **Cosine similarity query:** `1 - (embedding <=> $1::vector)` — pgvector operator, higher = more similar
- **Drizzle sql template:** Use `sql<number>` for the score column
- **Null filter:** Only search messages that have a non-null embedding
- **Schema:** `messages` table, `chats` join for chat name if needed (optional — senderName is on messages)
- **SEARCH_TOKEN** is already in config.ts (z.string().min(16))

## Tasks

### Task 1 — Create `src/routes/search.ts`

```typescript
import { z } from 'zod';
import { sql, isNotNull } from 'drizzle-orm';
import crypto from 'node:crypto';
import type { FastifyPluginAsyncZod } from 'fastify-type-provider-zod';
import { messages } from '../db/schema.js';

const searchResultSchema = z.object({
  id: z.string(),
  chatId: z.string(),
  senderName: z.string().nullable(),
  timestamp: z.date(),
  text: z.string().nullable(),
  score: z.number(),
});

const searchResponseSchema = z.object({
  results: z.array(searchResultSchema),
});

const searchQuerySchema = z.object({
  q: z.string().min(1).max(2000),
});

function makeSearchAuthHandler(token: string) {
  return async function searchAuthHandler(request: { headers: Record<string, string | string[] | undefined> }, reply: { send: (body: unknown) => Promise<void>; code: (code: number) => { send: (body: unknown) => Promise<void> } }) {
    const authHeader = (request.headers['authorization'] ?? '') as string;
    const prefix = 'Bearer ';
    if (!authHeader.startsWith(prefix)) {
      return reply.code(401).send({ error: 'Unauthorized' });
    }
    const provided = authHeader.slice(prefix.length);
    const tokenBuf = Buffer.from(token, 'utf8');
    const providedBuf = Buffer.from(provided, 'utf8');
    if (tokenBuf.byteLength !== providedBuf.byteLength || !crypto.timingSafeEqual(tokenBuf, providedBuf)) {
      return reply.code(401).send({ error: 'Unauthorized' });
    }
  };
}

const searchRoutes: FastifyPluginAsyncZod = async (fastify) => {
  const authHandler = makeSearchAuthHandler(fastify.config.SEARCH_TOKEN);

  fastify.get(
    '/search',
    {
      schema: {
        querystring: searchQuerySchema,
        response: { 200: searchResponseSchema, 401: z.object({ error: z.string() }) },
      },
      preHandler: [authHandler],
    },
    async (request) => {
      const { q } = request.query;

      // Embed the query using the same OpenAI client as enrich.ts
      const embeddingResult = await fastify.openai.embeddings.create({
        model: 'text-embedding-3-small',
        input: q,
      });
      const queryEmbedding = embeddingResult.data[0]!.embedding;
      const embeddingLiteral = `[${queryEmbedding.join(',')}]`;

      const results = await fastify.db
        .select({
          id: messages.id,
          chatId: messages.chatId,
          senderName: messages.senderName,
          timestamp: messages.timestamp,
          text: messages.text,
          score: sql<number>`1 - (embedding <=> ${embeddingLiteral}::vector)`,
        })
        .from(messages)
        .where(isNotNull(messages.embedding))
        .orderBy(sql`embedding <=> ${embeddingLiteral}::vector`)
        .limit(20);

      return { results };
    },
  );
};

export default searchRoutes;
```
